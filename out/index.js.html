/**
 * @route POST /users
 * @description Register a new user.
 * @param {Object} reqBody - User details.
 * @param {string} reqBody.Username - User's username.
 * @param {string} reqBody.UserPassword - User's password.
 * @param {string} reqBody.UserEmail - User's email.
 * @param {Date} [reqBody.UserBirthday] - User's birthday.
 * @returns {Object} Newly created user details.
 */
app.post(
  "/users",
  [
    check("Username", "Username is required").isLength({ min: 5 }),
    check(
      "Username",
      "Username contains non alphanumeric characters - not allowed."
    ).isAlphanumeric(),
    check("UserPassword", "Password is required").not().isEmpty(),
    check("UserEmail", "Email does not appear to be valid").isEmail(),
  ],
  async (req, res) => {
    //check the validation object for errors
    let errors = validationResult(req);
    if (!errors.isEmpty()) {
      return res.status(422).json({ errors: errors.array() });
    }
    let hashedPassword = Users.hashPassword(req.body.UserPassword);
    await Users.findOne({ Username: req.body.Username })
      //Search to see if a user with the resquested username already exists
      .then((user) => {
        console.log("Found user ", user);
        if (user) {
          //If the user is found, send a response that it already exists
          return res.status(400).send(req.body.Username + "already exists");
        } else {
          Users.create({
            Username: req.body.Username,
            UserPassword: hashedPassword,
            UserEmail: req.body.UserEmail,
            UserBirthday: req.body.UserBirthday,
          })
            .then((user) => {
              console.log("User from API: ", user);
              res.status(201).json(user);
            })
            .catch((error) => {
              console.error(error);
              res.status(500).send("Error: " + error);
            });
        }
      })
      .catch((error) => {
        console.error(error);
        res.status(500).send("Error: " + error);
      });
  }
);

/**
 * @route PUT /users/:Username
 * @description Update user information.
 * @param {string} Username - Username of the user.
 * @param {Object} reqBody - Updated user details.
 * @param {string} [reqBody.Username] - Updated username.
 * @param {string} [reqBody.UserPassword] - Updated password.
 * @param {string} [reqBody.UserEmail] - Updated email.
 * @param {Date} [reqBody.UserBirthday] - Updated birthday.
 * @returns {Object} Updated user details.
 */
app.put("/users/:Username", (req, res) => {
  Users.findOneAndUpdate(
    { Username: req.params.Username },
    {
      $set: {
        Username: req.body.Username,
        UserPassword: req.body.UserPassword,
        UserEmail: req.body.UserEmail,
        UserBirthday: req.body.UserBirthday,
      },
    },
    { new: true }
  )
    .then((updatedUser) => {
      res.json(updatedUser);
    })
    .catch((err) => {
      console.error(err);
      res.status(500).send("Error:" + err);
    });
});

/**
 * @route POST /users/:Username/favorites
 * @description Add a movie to a user's favorites by movie ID.
 * @param {string} Username - Username of the user.
 * @param {Object} reqBody - Movie details.
 * @param {string} reqBody._id - ID of the movie to add.
 * @returns {Object} Updated user details with favorite movies.
 */
app.post("/users/:Username/favorites", async (req, res) => {
  const { Username } = req.params;
  const { _id } = req.body; // Movie _id from the request body

  try {
    // Find the movie by _id in the Movie collection
    const movie = await Movies.findOne({ _id: _id });
    if (!movie) {
      console.log(_id);
      return res.status(404).send("Movie not found.");
    }

    /* Add the movie's _id to the User's UserFavoriteMovies array
    * Find the user by username and update
    */
    const user = await Users.findOneAndUpdate(
      { Username: Username },
      { $addToSet: { UserFavoriteMovies: _id } }, // $addToSet avoids duplicates
      { new: true }
    );
    if (!user) {
      return res.status(404).send("User not found.");
    }
    res.json(user);
  } catch (error) {
    console.error("Error adding movie to favorites:", error);
    res.status(500).send("Error adding movie to favorites");
  }
});

/**
 * @route DELETE /users/:Username/favorites
 * @description Remove a movie from a user's favorites by movie ID.
 * @param {string} Username - Username of the user.
 * @param {Object} reqBody - Movie details.
 * @param {string} reqBody._id - ID of the movie to remove.
 * @returns {Object} Updated user details with favorite movies removed.
 */
app.delete("/users/:Username/favorites", async (req, res) => {
  const { Username } = req.params;
  const { _id } = req.body; // Movie _id from the request body

  try {
    // Find the movie by _id in the Movie collection
    const movie = await Movies.findOne({ _id: _id });
    if (!movie) {
      console.log(_id);
      return res.status(404).send("Movie not found.");
    }

    // Remove the movie's _id from the User's UserFavoriteMovies array
    const user = await Users.findOneAndUpdate(
      { Username: Username },
      { $pull: { UserFavoriteMovies: _id } }, // $pull removes the movie's _id from the array
      { new: true }
    );
    if (!user) {
      return res.status(404).send("User not found.");
    }

    res.json(user); // Return the updated user data
  } catch (error) {
    console.error("Error removing movie from favorites:", error);
    res.status(500).send("Error removing movie from favorites");
  }
});
